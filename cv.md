## 💡 Code Example

```ts
// Finds "magic index" in an array (arr[i] === i)
const findMagic = (arr: number[]): number => {
  let flag = false;
  let result = 0;

  for (let i = 0; i < arr.length; i++) {
    if (arr[i] === i) {
      result = i;
      flag = true;
    }
  }

  return flag ? result : -1;
};
```
